API Reference

This page provides a reference for the main IPRO SDK APIs. The BLE API uses Zephyr-style bt_* callback-based APIs, and the HAL API provides low-level hardware access. For complete API documentation, please refer to the header file comments in the source code.

BLE Core Initialization

The BLE subsystem uses Zephyr-style bt_* APIs. Before calling bt_enable(), you must first call ipro_ble_ctlr_controller_init() and then hci_driver_ipro_ble_ctlr_init() in that order — otherwise bt_enable() will fail.

ipro_ble_controller.h / hci_driver_ipro_ble_ctlr.h 
/** Initialize the BLE controller (creates the controller task, must be called first)
 *  @param task_priority  FreeRTOS priority of the controller task
 *                        (the reference application uses configMAX_PRIORITIES - 1)
 */
int ipro_ble_ctlr_controller_init(unsigned int task_priority);

/** Initialize the HCI driver (must be called before bt_enable())
 *  @return 0 on success, negative error code on failure
 */
int hci_driver_ipro_ble_ctlr_init(void);
bluetooth/bluetooth.h 
/** Enable the Bluetooth subsystem
 *  @param cb  Completion callback, NULL for synchronous enable
 *  @return 0 on success, negative error code on failure
 */
int bt_enable(bt_ready_cb_t cb);

/** Set the Bluetooth device name
 *  @param name  New name string
 *  @return 0 on success, negative error code on failure
 */
int bt_set_name(const char *name);

/** Get the Bluetooth device name
 *  @return Device name string

Usage Example

BLE Initialization Example 
#include "freertos_app_hooks.h"
#include "ipro_log.h"
#include <FreeRTOS.h>
#include <task.h>

#include <zephyr/bluetooth/bluetooth.h>
#include <hci_driver_ipro_ble_ctlr.h>
#include <ipro_ble_controller.h>

#define TAG "APP"

static void bt_ready(int err)
{
    if (err) {
        IPRO_LOGE(TAG, "BLE init failed (err %d)", err);
        return;
    }
    IPRO_LOGI(TAG, "BLE initialized");
    /* Can start advertising or scanning */
}

static void app_task(void *arg)
{
    /* 1. Initialize the BLE controller (must run first) */
    int err = ipro_ble_ctlr_controller_init(configMAX_PRIORITIES - 1);
    if (err) {
        IPRO_LOGE(TAG, "Controller init failed (err %d)", err);
        vTaskDelete(NULL);
        return;
    }
    vTaskDelay(pdMS_TO_TICKS(50));  /* Give the controller task time to start */

    /* 2. Initialize the HCI driver */
    err = hci_driver_ipro_ble_ctlr_init();
    if (err == 0) {
        vTaskDelay(pdMS_TO_TICKS(50));
        /* 3. Enable the BLE Host */
        err = bt_enable(bt_ready);
    }
    if (err) {
        IPRO_LOGE(TAG, "BLE startup failed (err %d)", err);
    }
    vTaskDelete(NULL);
}

int main(void)
{
    ipro_platform_init();   /* heap + boot2 + log + shell */
    xTaskCreate(app_task, "app", 2048, NULL, 5, NULL);
    vTaskStartScheduler();
    while (1) {}
}

For the complete flow, see apps/bluetooth/ble/ipro_ble_remote (GATT + OTA reference application).

Service Definition Example

Custom GATT Service 
// Custom service UUID
static struct bt_uuid_128 my_svc_uuid = BT_UUID_INIT_128(
    0x01, 0x23, 0x45, 0x67, 0x89, 0xab, 0xcd, 0xef,
    0x01, 0x23, 0x45, 0x67, 0x89, 0xab, 0xcd, 0xef);

// Read callback
static ssize_t read_value(struct bt_conn *conn,
                          const struct bt_gatt_attr *attr,
                          void *buf, u16_t len, u16_t offset)
{
    return bt_gatt_attr_read(conn, attr, buf, len, offset,
                             &my_value, sizeof(my_value));
}

// Service definition
BT_GATT_SERVICE_DEFINE(my_svc,
    BT_GATT_PRIMARY_SERVICE(&my_svc_uuid),
    BT_GATT_CHARACTERISTIC(&my_char_uuid.uuid,
        BT_GATT_CHRC_READ | BT_GATT_CHRC_NOTIFY,
        BT_GATT_PERM_READ,
        read_value, NULL, NULL),
    BT_GATT_CCC(NULL, BT_GATT_PERM_READ | BT_GATT_PERM_WRITE),
);

 *  @param conn   Connection object
 *  @param handle Attribute handle

 *  @param length Data length

 *  @return 0 on success, negative error code on failure

    u16_t handle, const void *data, u16_t length, bool sign,
    bt_gatt_complete_func_t func, void *user_data);

Platform Initialization (freertos_app_hooks)

Header: components/os/freertos/freertos_app_hooks.h

All FreeRTOS applications share this component as the standard entry point. It also provides weak defaults for the FreeRTOS hooks (idle / tick / stack overflow / malloc failed), heap, and assert — an application can override any of them by defining a non-weak version in its own sources, without modifying the component.

freertos_app_hooks.h 
/** Standard platform init (call at the start of main())
 *  Runs in order: platform_init() → ipro_setup_heap() → hal_boot2_init()
 *                 → PSRAM check → ipro_log_init() → shell init (CONFIG_SHELL)
 *  After this returns, the app should create its tasks and call
 *  vTaskStartScheduler()
 */
void ipro_platform_init(void);

/** Set up FreeRTOS heap regions only
 *  Collects heap regions from linker symbols (OCRAM + optional PSRAM),
 *  sorts them by address, and calls vPortDefineHeapRegions().
 *  Safe to call once from main().
 */
void ipro_setup_heap(void);

Minimal Application Skeleton

main.c 
#include "freertos_app_hooks.h"
#include <FreeRTOS.h>
#include <task.h>

static void app_task(void *arg)
{
    for (;;) {
        /* Application logic */
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

int main(void)
{
    ipro_platform_init();   /* heap + boot2 + log + shell */
    xTaskCreate(app_task, "app", 1024, NULL, 5, NULL);
    vTaskStartScheduler();
    while (1) {}
}

FreeRTOS API

The SDK primarily uses FreeRTOS as the real-time operating system.

The BLE Host is Zephyr-based, but it runs on top of an internal FreeRTOS port layer; applications always use the FreeRTOS API — the Zephyr kernel API (k_thread_create, k_sleep, etc.) is neither needed nor available.
