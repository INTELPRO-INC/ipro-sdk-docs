教學 1：第一個 BLE 應用程式

學習如何使用 IPRO SDK 建立第一個 BLE 應用程式。本教學涵蓋 bt_enable() 初始化、廣播設定與連線回呼。

步驟 1：專案結構

每個 IPRO SDK 應用程式都是 apps/<category>/<name> 下的一個獨立目錄。建議複製 apps/bluetooth/ble/ipro_ble_remote/ 作為範本，並放在相同層級（例如 apps/bluetooth/ble/my_ble_app/）：

apps/bluetooth/ble/my_ble_app/
  .config            # Kconfig 設定（CONFIG_BLUETOOTH_LE_ENABLE=y 等）
  Makefile           # make 進入點（內含 PROJECT_NAME 與 IPRO_SDK_BASE）
  Kconfig            # 應用層 Kconfig 選項
  CMakeLists.txt     # 建構設定
  main.c             # 應用程式進入點

複製範本後，修改 Makefile 的 PROJECT_NAME 與 CMakeLists.txt 的 setup_project() 專案名稱。IPRO_SDK_BASE ?= $(abspath ../../../../) 依專案目錄深度調整。

步驟 2：.config 設定

在 .config 中啟用 BLE 功能。以下僅列關鍵設定，實務上請直接以 apps/bluetooth/ble/ipro_ble_remote/ 的 .config 為基礎修改：

# SoC 與工具鏈
CONFIG_IPRO7=y
CONFIG_CHIP="IPRO7"
CONFIG_CPU_ID="intelpro_ipro7"

# RTOS 選擇
CONFIG_FREERTOS_SUPPORT=y

# BLE Host + Controller 變體（prebuilt linkN library）
CONFIG_BLUETOOTH_LE_ENABLE=y
CONFIG_BT_CTLR_IPRO=y
CONFIG_BT_HOST_IPRO_V44=y
CONFIG_IPRO_BLE_CTLR_LIB="link1"
CONFIG_IPRO_BLE_CTLR_FEATURE="bt52"
CONFIG_IPRO_BLE_CTLR_CFG_DBG=n
CONFIG_BT_CONN=1
CONFIG_EM_SIZE=32

# NVDS: controller configuration storage; host bonds use BT_SETTINGS
# 缺少時會在連結階段出現 __nvds_start 未定義錯誤）
CONFIG_ENABLE_NVDS=y
CONFIG_CFG_NVDS=y

# Shell CLI（除錯用，可選）
CONFIG_SHELL=y

步驟 3：初始化平台與 BLE Controller

所有 FreeRTOS 應用程式共用 components/os/freertos/freertos_app_hooks.{c,h} 元件，它以 weak 函式提供 heap、FreeRTOS hooks 與平台初始化的預設實作。最小 main.c 只需呼叫 ipro_platform_init()，不需要手寫任何 hooks 或 heap 設定：

#include <stdio.h>
#include <FreeRTOS.h>
#include <task.h>
#include "ipro_ble_controller.h"          // ipro_ble_ctlr_controller_init()
#include "freertos_app_hooks.h"   // ipro_platform_init()

static void app_task(void *arg);  // 應用主迴圈（見步驟 4）

int main(void)
{
    // 1. 平台初始化（heap + boot2 + log + shell，由 freertos_app_hooks 提供）
    ipro_platform_init();

    // BLE 初始化由排程器啟動後的 app_task 執行。

    // 2. 建立應用 Task
    xTaskCreate(app_task, "app", 2048, NULL, 3, NULL);

    // 3. 啟動 FreeRTOS 排程器
    vTaskStartScheduler();
    return 0;
}

若需要客製某個 hook（例如 vApplicationIdleHook），在應用程式內定義非 weak 版本即可，連結器會自動採用，不要把 hooks/heap 樣板程式碼複製回 main.c。

步驟 4：啟用 BLE Host 與設定 Advertising

在應用 Task 中依序呼叫 hci_driver_ipro_ble_ctlr_init() 與 bt_enable()（順序固定：ipro_ble_ctlr_controller_init → hci_driver_ipro_ble_ctlr_init → bt_enable），並在 ready 回呼中啟動廣播：

#include <zephyr/bluetooth/bluetooth.h>
#include <zephyr/bluetooth/conn.h>
#include <hci_driver_ipro_ble_ctlr.h>

static const struct bt_data ad[] = {
    BT_DATA_BYTES(BT_DATA_FLAGS,
                  BT_LE_AD_GENERAL | BT_LE_AD_NO_BREDR),
    BT_DATA_BYTES(BT_DATA_NAME_COMPLETE, 'I','P','R','O'),
};

static void bt_ready(int err)
{
    if (err) {
        printf("BLE init failed (err %d)\n", err);
        return;
    }
    // 啟動廣播
    err = bt_le_adv_start(BT_LE_ADV_CONN_FAST_1, ad, ARRAY_SIZE(ad),
                          NULL, 0);
    if (err) {
        printf("Advertising failed (err %d)\n", err);
    }
}

static void app_task(void *arg)
{
    // HCI driver 必須在 bt_enable() 之前初始化
    int err = ipro_ble_ctlr_controller_init(configMAX_PRIORITIES - 1);
    if (err) {
        printf("Controller init failed (err %d)\n", err);
        vTaskDelete(NULL);
        return;
    }
    vTaskDelay(pdMS_TO_TICKS(50));
    err = hci_driver_ipro_ble_ctlr_init();
    if (err) {
        printf("HCI driver init failed (err %d)\n", err);
        vTaskDelete(NULL);
        return;
    }

    vTaskDelay(pdMS_TO_TICKS(50));
    err = bt_enable(bt_ready);
    if (err) {
        printf("bt_enable failed (err %d)\n", err);
    }

    for (;;) {
        vTaskDelay(portMAX_DELAY);
    }
}

步驟 5：連線回呼

定義 struct bt_conn_cb 並用 bt_conn_cb_register() 註冊（建議放在 bt_ready() 開頭）：

static void connected(struct bt_conn *conn, uint8_t err)
{
    if (err) {
        printf("Connection failed (err 0x%02x)\n", err);
        return;
    }
    printf("Connected\n");
}

static void disconnected(struct bt_conn *conn, uint8_t reason)
{
    printf("Disconnected (reason 0x%02x)\n", reason);
    // 重新啟動廣播
    bt_le_adv_start(BT_LE_ADV_CONN_FAST_1, ad, ARRAY_SIZE(ad),
                    NULL, 0);
}

static struct bt_conn_cb conn_callbacks = {
    .connected = connected,
    .disconnected = disconnected,
};

// 在 bt_ready() 中註冊：
//     bt_conn_cb_register(&conn_callbacks);

步驟 6：建構與燒錄

make -C apps/bluetooth/ble/my_ble_app
# → apps/bluetooth/ble/my_ble_app/build/build_out/my_ble_app_IPRO7.bin

# 燒錄（序列埠名稱依實際裝置而定）

    --chipname ipro7 \
    --port /dev/cu.usbserial-XXXX \
    --firmware apps/bluetooth/ble/my_ble_app/build/build_out/my_ble_app_IPRO7.bin \

# 或用 build_freertos.sh 一次完成建構＋燒錄＋監看

教學 2：建構與燒錄流程

掌握 IPRO SDK 的建構系統和韌體燒錄流程。

建構方式

IPRO SDK 提供兩種建構方式：

# 方式 1：使用 build_freertos.sh（推薦，可串接多個 action）
./build_freertos.sh ipro_ble_remote build

# 串接 actions：建構 + 燒錄 + 開啟序列埠監看
./build_freertos.sh ipro_ble_remote build flash monitor

# 其他 actions：clean / size（ELF 大小與記憶體區段用量報告）
./build_freertos.sh clean build bluetooth/ble/ipro_ble_remote
./build_freertos.sh size bluetooth/ble/ipro_ble_remote

# 列出所有可建構的專案
./build_freertos.sh -l

# 方式 2：使用 Make（底層方式，需用完整巢狀路徑 apps/<category>/<name>）
make -C apps/bluetooth/ble/ipro_ble_remote

# 清除編譯
make -C apps/bluetooth/ble/ipro_ble_remote clean

build_freertos.sh 常用選項：-p 指定序列埠、-t 設定逾時秒數、-w 設定 monitor 等待的輸出樣式（regex）。建構產出位於 apps/<category>/<name>/build/build_out/，二進位檔名包含晶片後綴，例如 ipro_ble_remote_IPRO7.bin。

.config 設定說明

每個應用程式都有自己的 .config 檔案，使用 Linux Kernel Kconfig 格式。以 BLE 應用為例：

# SoC 選擇
CONFIG_IPRO7=y
CONFIG_CHIP="IPRO7"
CONFIG_CPU_ID="intelpro_ipro7"

# RTOS 選擇
CONFIG_FREERTOS_SUPPORT=y

# BLE 功能與 Controller 變體
CONFIG_BLUETOOTH_LE_ENABLE=y
CONFIG_BT_CTLR_IPRO=y
CONFIG_BT_HOST_IPRO_V44=y
CONFIG_IPRO_BLE_CTLR_LIB="link1"
CONFIG_IPRO_BLE_CTLR_FEATURE="bt52"
CONFIG_IPRO_BLE_CTLR_CFG_DBG=n
CONFIG_BT_CONN=1
CONFIG_EM_SIZE=32

# NVDS: controller configuration storage; host bonds use BT_SETTINGS
CONFIG_ENABLE_NVDS=y
CONFIG_CFG_NVDS=y

# Shell CLI
CONFIG_SHELL=y

燒錄韌體

使用 另行安裝的燒錄工具 ipro_iot_tool_lite（序列埠名稱依實際裝置而定，macOS 可用 ls /dev/cu.usbserial-* 查詢）：

# 基本燒錄指令
/path/to/ipro_iot_tool_lite \
    --chipname ipro7 \
    --port /dev/cu.usbserial-XXXX \
    --firmware apps/bluetooth/ble/ipro_ble_remote/build/build_out/ipro_ble_remote_IPRO7.bin \
    --baudrate 921600 \
    --flash-pin 0x02

macOS 序列埠注意事項

在 macOS 上必須使用 /dev/cu.* 而非 /dev/tty.*。tty.* 裝置會等待 DCD 信號而無法正常使用。

序列埠監控

V2.0.1 未附舊版監看腳本，可另行安裝 pyserial 並開啟終端：

# 另行安裝；埠名與鮑率請依韌體設定調整
python3 -m pip install pyserial
python3 -m serial.tools.miniterm /dev/cu.usbserial-XXXX 115200

自動化流程

build_freertos.sh 可將建構、燒錄、監看串成一條指令，搭配 -w/-t 等待預期輸出，適合整合進自動化指令稿：

# 一行完成：建構 + 燒錄 + 監看
./build_freertos.sh ipro_ble_remote build flash monitor

# 等待應用開機輸出（60 秒逾時）
./build_freertos.sh -w "BLE 5.4 stack initialized" -t 60 \
    ipro_ble_remote build flash monitor

教學 3：BLE 吞吐量測試

使用 ipro_throughput_test 應用程式測試 BLE 資料傳輸吞吐量，2M PHY 實測可達 999 Kbps。

概述

吞吐量測試在兩塊 IPRO7 開發板之間進行，一塊作為 Slave（Peripheral），另一塊作為 Master（Central）。應用程式使用原生 Zephyr bt_* API 實作。

步驟 1：建構與燒錄

將相同的韌體燒錄到兩塊開發板（兩塊板子的序列埠名稱不同，請依實際裝置調整）：

# 建構吞吐量測試應用
make -C apps/bluetooth/ble/ipro_throughput_test

# 燒錄到兩塊板子
/path/to/ipro_iot_tool_lite \
    --chipname ipro7 --port /dev/cu.usbserial-XXXX \
    --firmware apps/bluetooth/ble/ipro_throughput_test/build/build_out/ipro_throughput_test_IPRO7.bin \
    --baudrate 921600 --flash-pin 0x02

步驟 2：啟動測試

# 在 Slave 板上執行：
tp slave

# 在 Master 板上執行（自動掃描連線）：
tp master

# 或指定 MAC 位址連線：
tp master XX:XX:XX:XX:XX:XX

步驟 3：執行吞吐量測試

# 連線成功後，先切換到 2M PHY（達到 999 Kbps 的必要條件）
tp phy 2

# 開始測試（預設 200 bytes/packet, 10 秒）
tp start

# 自訂封包大小和持續時間（最高吞吐量使用 244 bytes）
tp start 244 10    # 244 bytes, 10 秒

# 查看即時統計
tp stats

# 停止測試
tp stop

其他調校指令：tp phy <1|2|3> 切換 PHY（1=1M, 2=2M, 3=Coded）、tp conn <min> <max> 調整連線間隔（單位 1.25ms）、tp interval <ms> 調整 TX 間隔。

效能優化要點

IPRO SDK 通過以下優化達到 999 Kbps 吞吐量：

  • 連線參數調整：最小連線間隔、零 Slave Latency
  • MTU 協商：最大 MTU 以減少封包開銷
  • TX Descriptor 優化：BLE_NB_TX_DESC_PER_CON 參數調整
  • DLE（Data Length Extension）：啟用最大傳輸長度
  • 2M PHY：使用 BLE 2M PHY 提升傳輸速率

實測參考數據（244 bytes/packet）：

PHY連線間隔吞吐量
1M7.5 ms507 Kbps
2M7.5 ms999 Kbps
2M15 ms984 Kbps

預期結果

測試期間每秒輸出一行即時統計，tp stats 顯示累計結果（以下數值為示意，實際依 PHY 與連線參數而異）：

[TP] TX: 512 pkts, 124928 bytes, 998.4 Kbps | RX: 0 pkts, 0 bytes, 0.0 Kbps | Err: 0

TX: 5016 packets, 1223904 bytes, 999.15 Kbps (124.89 KBps)
RX: 0 packets, 0 bytes, 0.00 Kbps (0.00 KBps)

教學 4：LE Audio 入門

了解 BLE 5.4 LE Audio 的核心概念，包括 BAP/BIS 架構，以及如何使用 IPRO SDK 的 LE Audio 範例應用程式。

LE Audio 核心概念

  • BAP (Basic Audio Profile)：定義音訊串流的建立與管理
  • CIS (Connected Isochronous Stream)：點對點音訊傳輸，用於耳機等裝置
  • BIS (Broadcast Isochronous Stream)：一對多廣播音訊，用於 Auracast 場景
  • LC3 Codec：LE Audio 標準編解碼器，低延遲高品質
  • GAF (Generic Audio Framework)：統一的音訊框架，整合 BAP、VCP、MCP、CAP 等 Profile

Controller 變體選擇

LIBACL角色
link1_periph1Peripheral
link11Central + Peripheral
link22Central + Peripheral
link44Central + Peripheral

基本 BLE 選擇 CONFIG_IPRO_BLE_CTLR_FEATURE="bt52"；BLE 5.4 與兩路 ISO 串流（CIS/BIS）選擇 "bt54_iso2"，後者要求 CONFIG_EM_SIZE 至少 32 KB。Host 連線上限不可超過控制器容量；請保留各應用隨附配置。

BIS 廣播範例

V2.0.1 未隨附獨立 ipro_le_audio_bis_source 與 ipro_le_audio_bis_sink 應用。Host 包含廣播 API；完整廣播參考應用請聯絡原廠取得。

CIS Unicast 範例 (Headset)

使用 ipro_le_audio_headset 作為 LE Audio 耳機：

# 建構 LE Audio Headset 範例
make -C apps/bluetooth/le_audio/ipro_le_audio_headset

.config 設定重點

CONFIG_BLUETOOTH_LE_ENABLE=y
CONFIG_BT_CTLR_IPRO=y
CONFIG_BT_HOST_IPRO_V44=y
CONFIG_IPRO_BLE_CTLR_LIB="link1_periph"
CONFIG_IPRO_BLE_CTLR_FEATURE="bt54_iso2"
CONFIG_IPRO_BLE_CTLR_CFG_DBG=n
CONFIG_BT_CONN=1
CONFIG_EM_SIZE=32
CONFIG_BT_AUDIO=y
CONFIG_BT_BAP_UNICAST_SERVER=y
CONFIG_BT_ASCS=y
CONFIG_BT_PACS=y
CONFIG_COMPONENTS_LC3_ENABLE=y
