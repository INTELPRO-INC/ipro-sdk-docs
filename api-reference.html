API 參考

本頁提供 IPRO SDK 主要 API 的參考文件。BLE API 使用 Zephyr 風格的 bt_* 回呼式 API，HAL API 提供底層硬體存取。完整的 API 文件請參考原始碼中的標頭檔註解。

本頁的 HAL 與 RTOS API 適用於 FreeRTOS 應用。V2.1.0 起以 Zephyr RTOS 建構時，周邊請改用 Zephyr 的 device driver API（底層為相同的 HAL 與 SoC 驅動），請見 Zephyr RTOS。

以下初始化流程適用於 IPRO7 的 FreeRTOS 應用（V2.1.0 僅支援 IPRO7）；Zephyr 的 BLE 範例請見 Zephyr RTOS。IPRO6S 的 BTDM 初始化請參考 V2.0.1 的 apps/platform/ipro6_demo/。

BLE 核心初始化

BLE 子系統使用 Zephyr 風格的 bt_* API。呼叫 bt_enable() 之前，必須先依序完成 ipro_ble_ctlr_controller_init() 與 hci_driver_ipro_ble_ctlr_init()，否則 bt_enable() 會失敗。

ipro_ble_controller.h / hci_driver_ipro_ble_ctlr.h 
/** 初始化 BLE Controller（建立 controller 任務，必須最先呼叫）
 *  @param task_priority  Controller 任務的 FreeRTOS 優先權
 *                        （參考應用使用 configMAX_PRIORITIES - 1）
 */
int ipro_ble_ctlr_controller_init(unsigned int task_priority);

/** 初始化 HCI driver（必須在 bt_enable() 之前呼叫）
 *  @return 0 成功，負值錯誤碼
 */
int hci_driver_ipro_ble_ctlr_init(void);
bluetooth/bluetooth.h 
/** 啟用藍牙子系統
 *  @param cb  完成回呼函式，NULL 表示同步啟用
 *  @return 0 成功，負值錯誤碼
 */
int bt_enable(bt_ready_cb_t cb);

/** 設定藍牙裝置名稱
 *  @param name  新名稱字串
 *  @return 0 成功，負值錯誤碼
 */
int bt_set_name(const char *name);

/** 取得藍牙裝置名稱
 *  @return 裝置名稱字串
 */
const char *bt_get_name(void);

使用範例

BLE 初始化範例 
#include "freertos_app_hooks.h"
#include "ipro_log.h"
#include <FreeRTOS.h>
#include <task.h>

#include <zephyr/bluetooth/bluetooth.h>
#if defined(CONFIG_BT_SETTINGS)
#include <zephyr/settings/settings.h>
#endif
#include <hci_driver_ipro_ble_ctlr.h>
#include <ipro_ble_controller.h>

#define TAG "APP"

static void bt_ready(int err)
{
    if (err) {
        IPRO_LOGE(TAG, "BLE init failed (err %d)", err);
        return;
    }
#if defined(CONFIG_BT_SETTINGS)
    err = settings_load_subtree("bt");
    if (err) {
        IPRO_LOGE(TAG, "Bluetooth settings load failed (err %d)", err);
        return;
    }
#endif
    IPRO_LOGI(TAG, "BLE initialized");
    /* 可以開始廣播或掃描 */
}

static void app_task(void *arg)
{
    /* 1. 初始化 BLE Controller（必須最先執行） */
    int err = ipro_ble_ctlr_controller_init(configMAX_PRIORITIES - 1);
    if (err) {
        IPRO_LOGE(TAG, "Controller init failed (err %d)", err);
        vTaskDelete(NULL);
        return;
    }
    vTaskDelay(pdMS_TO_TICKS(50));  /* 等待 controller 任務啟動 */

    /* 2. 初始化 HCI driver */
    err = hci_driver_ipro_ble_ctlr_init();
    if (err == 0) {
        vTaskDelay(pdMS_TO_TICKS(50));
        /* 3. 啟用 BLE Host */
        err = bt_enable(bt_ready);
    }
    if (err) {
        IPRO_LOGE(TAG, "BLE startup failed (err %d)", err);
    }
    vTaskDelete(NULL);
}

int main(void)
{
    ipro_platform_init();   /* heap + boot2 + log + shell */
    xTaskCreate(app_task, "app", 2048, NULL, 5, NULL);
    vTaskStartScheduler();
    while (1) {}
}

服務定義範例

自訂 GATT 服務 
// 自訂服務 UUID
static uint8_t my_value;
static struct bt_uuid_16 my_char_uuid = BT_UUID_INIT_16(0xfff1);

static struct bt_uuid_128 my_svc_uuid = BT_UUID_INIT_128(
    0x01, 0x23, 0x45, 0x67, 0x89, 0xab, 0xcd, 0xef,
    0x01, 0x23, 0x45, 0x67, 0x89, 0xab, 0xcd, 0xef);

// 讀取回呼
static ssize_t read_value(struct bt_conn *conn,
                          const struct bt_gatt_attr *attr,
                          void *buf, uint16_t len, uint16_t offset)
{
    return bt_gatt_attr_read(conn, attr, buf, len, offset,
                             &my_value, sizeof(my_value));
}

// 服務定義
BT_GATT_SERVICE_DEFINE(my_svc,
    BT_GATT_PRIMARY_SERVICE(&my_svc_uuid),
    BT_GATT_CHARACTERISTIC(&my_char_uuid.uuid,
        BT_GATT_CHRC_READ | BT_GATT_CHRC_NOTIFY,
        BT_GATT_PERM_READ,
        read_value, NULL, NULL),
    BT_GATT_CCC(NULL, BT_GATT_PERM_READ | BT_GATT_PERM_WRITE),
);

平台初始化 (freertos_app_hooks)

標頭檔：components/os/freertos/freertos_app_hooks.h

所有 FreeRTOS 應用共用此元件作為標準進入點。它同時以 weak 預設提供 FreeRTOS hooks（idle / tick / stack overflow / malloc failed）與 heap、assert —— 應用只需在自己的原始碼定義非 weak 版本即可覆寫，無需修改元件。

freertos_app_hooks.h 
/** 標準平台初始化（在 main() 開頭呼叫）
 *  依序執行：platform_init() → ipro_setup_heap() → hal_boot2_init()
 *           → PSRAM 檢查 → ipro_log_init() → shell 初始化 (CONFIG_SHELL)
 *  返回後應用即可建立任務並呼叫 vTaskStartScheduler()
 */
void ipro_platform_init(void);

/** 僅設定 FreeRTOS heap 區域
 *  從 linker symbols 收集 heap 區域（OCRAM + 可選 PSRAM），
 *  排序後呼叫 vPortDefineHeapRegions()。只能在 main() 呼叫一次。
 */
void ipro_setup_heap(void);

最小應用骨架

main.c 
#include "freertos_app_hooks.h"
#include <FreeRTOS.h>
#include <task.h>

static void app_task(void *arg)
{
    for (;;) {
        /* 應用邏輯 */
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

int main(void)
{
    ipro_platform_init();   /* heap + boot2 + log + shell */
    xTaskCreate(app_task, "app", 1024, NULL, 5, NULL);
    vTaskStartScheduler();
    while (1) {}
}

FreeRTOS API

SDK 主要使用 FreeRTOS 作為即時作業系統。

BLE Host 的 Zephyr 相容層由 FreeRTOS port 支援。應用任務、延遲與同步請使用 FreeRTOS API；相容層只提供 Host 移植所需功能，不是完整 Zephyr kernel。需要完整 Zephyr kernel 時，請改用 V2.1.0 起提供的 Zephyr RTOS 路徑。
